# Problem: https://leetcode.com/problems/count-commas-in-range/
# Leetcode Problem No: 3870
# Relevance: Math

class Solution:

    # Logic: Mathematical Threshold Counting (O(1) Time, O(1) Space)
    # 1. In standard numerical representation, commas are placed every three digits from the right:
    #    - Numbers from 1 to 999 contain 0 commas.
    #    - Numbers from 1,000 to 99,999 (within the problem constraint n <= 10^5) contain exactly 1 comma.
    # 2. Every number >= 1,000 up to n contributes exactly 1 comma to the total count.
    # 3. Therefore, the total number of commas is simply the number of integers in the range [1000, n].
    #    - If n >= 1000: count = n - 1000 + 1 = n - 999 = n - (10^3 - 1).
    #    - If n < 1000: count = 0.
    # 4. Result: `max(0, n - (10^3 - 1))`.
    #
    # Time Complexity: O(1) direct mathematical calculation.
    # Space Complexity: O(1) auxiliary space.
    #
    # Dry run for n = 1002:
    # - n >= 1000: 1002 - 999 = 3 (for numbers 1,000, 1,001, 1,002).
    # - Returns 3.
    #
    # Dry run for n = 999:
    # - n < 1000: max(0, 999 - 999) = 0.
    # - Returns 0.

    def countCommas(self, n: int) -> int:
        return max(0, n - (10**3 - 1))

if __name__ == "__main__":
    sol = Solution()

    # Example 1: n = 1002 -> Expected: 3
    n1 = 1002
    print(f"Result for n={n1}: {sol.countCommas(n1)}")

    # Example 2: n = 999 -> Expected: 0
    n2 = 999
    print(f"Result for n={n2}: {sol.countCommas(n2)}")

    # Example 3: n = 50 -> Expected: 0
    n3 = 50
    print(f"Result for n={n3}: {sol.countCommas(n3)}")
