# Problem: https://leetcode.com/problems/maximum-alternating-sum-of-squares/
# Leetcode Problem No: 3727
# Relevance: Array, Greedy, Sorting

from typing import List

class Solution:

    # Logic: Greedy Sorting and Optimal Partitioning (O(N log N) Time, O(1) Auxiliary Space)
    # 1. The alternating score of an arrangement is:
    #    arr[0]^2 - arr[1]^2 + arr[2]^2 - arr[3]^2 + ...
    # 2. Key Observations:
    #    - Squaring eliminates signs: negative and positive numbers contribute equally to their square.
    #    - In an array of size `n`, there are `ceil(n / 2)` positive slots (even indices: 0, 2, 4...)
    #      and `floor(n / 2) = n // 2` negative slots (odd indices: 1, 3, 5...).
    # 3. Greedy Strategy:
    #    - To maximize the alternating score, we must assign the `n // 2` smallest squared values to the negative slots,
    #      and the remaining larger squared values to the positive slots.
    #    - Replace each number with its square in-place, sort the array in ascending order.
    #    - Subtract the sum of the first `n // 2` elements and add the sum of the remaining `n - n // 2` elements.
    #
    # Time Complexity: O(N log N) where N is the length of `nums`, dominated by sorting.
    # Space Complexity: O(1) auxiliary space (modifying and sorting `nums` in-place).
    #
    # Dry run for nums = [2, 1, -3, 2]:
    # - n = 4, n // 2 = 2
    # - Squaring: nums = [4, 1, 9, 4]
    # - Sorted: nums = [1, 4, 4, 9]
    # - Smallest 2 elements (negative slots): [1, 4] -> sum = 5
    # - Remaining elements (positive slots): [4, 9] -> sum = 13
    # - Return: -5 + 13 = 8.

    def maxAlternatingSum(self, nums: List[int]) -> int:
        n = len(nums)
        for i in range(n):
            nums[i] = nums[i] * nums[i]
        nums.sort()
        return -sum(nums[:n // 2]) + sum(nums[n // 2:])

if __name__ == "__main__":
    sol = Solution()

    # Example 1: nums = [1, 2, 3] -> Expected: 12
    nums1 = [1, 2, 3]
    print(f"Result for {nums1}: {sol.maxAlternatingSum(nums1)}")

    # Example 2: nums = [2, 1, -3, 2] -> Expected: 8
    nums2 = [2, 1, -3, 2]
    print(f"Result for {nums2}: {sol.maxAlternatingSum(nums2)}")
