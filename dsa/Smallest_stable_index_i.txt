# Problem: https://leetcode.com/problems/smallest-stable-index-i/
# Leetcode Problem No: 3903
# Relevance: Array, Prefix Max, Suffix Min

from typing import List

class Solution:

    # Logic: Prefix Maximum & Suffix Minimum Precomputation (O(N) Time, O(N) Space)
    # 1. An index `i` is stable if: `max(nums[0...i]) - min(nums[i...n-1]) <= k`.
    # 2. We can precompute:
    #    - `maxArr[i]`: the maximum value from index 0 to index `i` (Prefix Max).
    #    - `minArr[i]`: the minimum value from index `i` to index `n-1` (Suffix Min).
    # 3. We iterate from index 0 to `n-1`:
    #    - Compute `val = maxArr[i] - minArr[i]`.
    #    - If `val <= k`, return `i` immediately as it is the smallest valid stable index.
    # 4. If no such index satisfies the condition, return `-1`.
    #
    # Time Complexity: O(N) where N is the length of `nums`.
    # Space Complexity: O(N) for storing the prefix max and suffix min arrays.
    #
    # Dry run for nums = [3, 2, 1, 4, 5], k = 3:
    # - maxArr = [3, 3, 3, 4, 5]
    # - minArr = [1, 1, 1, 4, 5]
    # - i = 0: maxArr[0] - minArr[0] = 3 - 1 = 2 <= 3 -> Valid!
    # - Returns 0.
    #
    # Dry run for nums = [10, 8, 4, 6], k = 2:
    # - maxArr = [10, 10, 10, 10]
    # - minArr = [4, 4, 4, 6]
    # - i = 0: 10 - 4 = 6 > 2
    # - i = 1: 10 - 4 = 6 > 2
    # - i = 2: 10 - 4 = 6 > 2
    # - i = 3: 10 - 6 = 4 > 2
    # - Returns -1.

    def firstStableIndex(self, nums: List[int], k: int) -> int:
        n = len(nums)
        maxArr = [-1] * n
        minArr = [-1] * n

        maxArr[0] = nums[0]
        for i in range(1, n):
            maxArr[i] = max(maxArr[i - 1], nums[i])

        minArr[-1] = nums[-1]
        for i in range(n - 2, -1, -1):
            minArr[i] = min(minArr[i + 1], nums[i])

        for i in range(n):
            val = maxArr[i] - minArr[i]
            if val <= k:
                return i

        return -1

if __name__ == "__main__":
    sol = Solution()

    # Example 1: nums = [3, 2, 1, 4, 5], k = 3 -> Expected: 0
    nums1 = [3, 2, 1, 4, 5]
    k1 = 3
    print(f"Result for {nums1} (k={k1}): {sol.firstStableIndex(nums1, k1)}")

    # Example 2: nums = [10, 8, 4, 6], k = 2 -> Expected: -1
    nums2 = [10, 8, 4, 6]
    k2 = 2
    print(f"Result for {nums2} (k={k2}): {sol.firstStableIndex(nums2, k2)}")

    # Example 3: nums = [5, 1, 3, 2, 4], k = 1 -> Expected: 4
    nums3 = [5, 1, 3, 2, 4]
    k3 = 1
    print(f"Result for {nums3} (k={k3}): {sol.firstStableIndex(nums3, k3)}")
