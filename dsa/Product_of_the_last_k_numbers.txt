# Problem: https://leetcode.com/problems/product-of-the-last-k-numbers/
# Leetcode Problem No: 1352
# Relevance: Array, Math, Design, Data Stream, Prefix Sum

class ProductOfNumbers:

    # Logic: Prefix Product with Zero-Reset (O(1) Time for add & getProduct, O(N) Space)
    # 1. We maintain a running prefix product array `pref = [1]`.
    # 2. Handling Numbers (`add`):
    #    - If `num == 0`:
    #      Any query requesting `k` numbers that spans across or includes this zero will result in 0.
    #      We can simply reset `pref = [1]`. Any subsequent query with `k >= len(pref)` knows that a zero
    #      occurred within the last `k` numbers.
    #    - If `num > 0`:
    #      Append `pref[-1] * num` to `pref`.
    # 3. Computing Product (`getProduct`):
    #    - If `k >= len(pref)`: A zero was added within the last `k` numbers, so return 0.
    #    - Otherwise: The product of the last `k` numbers is given by:
    #      `pref[-1] // pref[-1 - k]`.
    #
    # Time Complexity:
    # - add(num): O(1) amortized to append to the list.
    # - getProduct(k): O(1) direct list index and integer division.
    # Space Complexity: O(N) where N is the number of elements added since the last zero.
    #
    # Dry run:
    # - add(3): pref = [1, 3]
    # - add(0): pref = [1] (reset)
    # - add(2): pref = [1, 2]
    # - add(5): pref = [1, 2, 10]
    # - add(4): pref = [1, 2, 10, 40]
    # - getProduct(2): k=2 < 4 -> pref[-1] // pref[-1 - 2] = 40 // 2 = 20 (5 * 4 = 20)
    # - getProduct(3): k=3 < 4 -> pref[-1] // pref[-1 - 3] = 40 // 1 = 40 (2 * 5 * 4 = 40)
    # - getProduct(4): k=4 >= len(pref) (4) -> returns 0 (spans across the 0).

    def __init__(self):
        self.pref = [1]

    def add(self, num: int) -> None:
        if num == 0:
            self.pref = [1]
        else:
            self.pref.append(self.pref[-1] * num)

    def getProduct(self, k: int) -> int:
        if k >= len(self.pref):
            return 0
        return self.pref[-1] // self.pref[-1 - k]

if __name__ == "__main__":
    obj = ProductOfNumbers()
    obj.add(3)        # [3]
    obj.add(0)        # [3, 0]
    obj.add(2)        # [3, 0, 2]
    obj.add(5)        # [3, 0, 2, 5]
    obj.add(4)        # [3, 0, 2, 5, 4]

    print(f"getProduct(2): {obj.getProduct(2)}")  # Expected: 20 (5 * 4)
    print(f"getProduct(3): {obj.getProduct(3)}")  # Expected: 40 (2 * 5 * 4)
    print(f"getProduct(4): {obj.getProduct(4)}")  # Expected: 0  (spans the 0)

    obj.add(8)        # [3, 0, 2, 5, 4, 8]
    print(f"getProduct(2): {obj.getProduct(2)}")  # Expected: 32 (4 * 8)
