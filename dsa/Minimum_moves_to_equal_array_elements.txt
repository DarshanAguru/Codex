# Problem: https://leetcode.com/problems/minimum-moves-to-equal-array-elements/
# Leetcode Problem No: 453
# Relevance: Array, Math

from typing import List

class Solution:

    # Logic: Mathematical Equivalence via Inversion (O(N) Time, O(1) Space)
    # 1. In one move, we can increment n - 1 elements by 1.
    # 2. Key Mathematical Insight:
    #    - Incrementing n - 1 elements by 1 is mathematically equivalent to decrementing 1 element by 1
    #      in terms of the relative differences between elements.
    #    - Therefore, instead of increasing n - 1 elements to reach an ever-growing maximum, we can think of
    #      the operation as decrementing each element until it reaches the minimum element of the array.
    # 3. Derivation:
    #    - Let `min_val` be min(nums) and `n` be len(nums).
    #    - The number of moves required for element `nums[i]` to reach `min_val` is `nums[i] - min_val`.
    #    - Total moves = sum(nums[i] - min_val for all i)
    #                  = sum(nums) - sum(min_val for all i)
    #                  = sum(nums) - min(nums) * len(nums).
    #
    # Time Complexity: O(N) where N is the length of `nums`, scanning the array once to compute sum and min.
    # Space Complexity: O(1) auxiliary space.
    #
    # Dry run for nums = [1, 2, 3]:
    # - min(nums) = 1, len(nums) = 3, sum(nums) = 6
    # - Moves = 6 - (1 * 3) = 3
    # - Trace:
    #   Move 1: [2, 3, 3] (increment 1 and 2)
    #   Move 2: [3, 4, 3] (increment 2 and 3) -> wait, increment 1st and 3rd -> [3, 3, 4]
    #   Move 3: [4, 4, 4] (increment 1st and 2nd) -> 3 moves total.

    def minMoves(self, nums: List[int]) -> int:
        return sum(nums) - min(nums) * len(nums)

if __name__ == "__main__":
    sol = Solution()

    # Example 1: nums = [1, 2, 3] -> Expected: 3
    nums1 = [1, 2, 3]
    print(f"Result for {nums1}: {sol.minMoves(nums1)}")

    # Example 2: nums = [1, 1, 1] -> Expected: 0
    nums2 = [1, 1, 1]
    print(f"Result for {nums2}: {sol.minMoves(nums2)}")

    # Example 3: nums = [1, 10, 1] -> Expected: 9
    nums3 = [1, 10, 1]
    print(f"Result for {nums3}: {sol.minMoves(nums3)}")
