# Problem: https://leetcode.com/problems/minimum-positive-sum-subarray/
# Leetcode Problem No: 3364
# Relevance: Array, Sliding Window, Prefix Sum

from typing import List

class Solution:

    # Logic: Multi-Length Fixed Sliding Window (O((R - L + 1) * N) Time, O(1) Auxiliary Space)
    # 1. We are given an integer array `nums` and integers `l` and `r`.
    #    We need to find the minimum sum of a subarray whose length `k` satisfies `l <= k <= r`
    #    and whose sum is strictly positive (`sum > 0`).
    # 2. Sliding Window Strategy:
    #    - Iterate over each valid subarray length `k` in range `[l, r]`.
    #    - For each length `k`, slide a fixed-size window of size `k` over `nums`:
    #      - Add the incoming element `nums[j]`.
    #      - Remove the outgoing element `nums[j - k]` once `j >= k`.
    #      - When the window reaches full size (`j >= k - 1`), check if `0 < summ < res`.
    #        If so, update `res = summ`.
    # 3. If no such positive subarray is found (`res == float('inf')`), return -1; otherwise, return `res`.
    #
    # Time Complexity: O((r - l + 1) * N) where N is the length of `nums`.
    #                  Since N <= 100, this takes at most ~10^4 operations and runs in a few milliseconds.
    # Space Complexity: O(1) auxiliary space.
    #
    # Dry run for nums = [3, -2, 1, 4], l = 2, r = 3:
    # - Length k = 2:
    #   - Window [3, -2]: sum = 1 > 0 -> res = 1
    #   - Window [-2, 1]: sum = -1 <= 0
    #   - Window [1, 4]: sum = 5 > 0 -> min(1, 5) = 1
    # - Length k = 3:
    #   - Window [3, -2, 1]: sum = 2 > 0 -> min(1, 2) = 1
    #   - Window [-2, 1, 4]: sum = 3 > 0 -> min(1, 3) = 1
    # - Return: 1.

    def minimumSumSubarray(self, nums: List[int], l: int, r: int) -> int:
        n = len(nums)
        res = float("inf")

        for k in range(l, r + 1):
            summ = 0
            for j in range(n):
                summ += nums[j]
                if j >= k:
                    summ -= nums[j - k]
                if j >= k - 1 and 0 < summ < res:
                    res = summ

        return -1 if res == float("inf") else res

if __name__ == "__main__":
    sol = Solution()

    # Example 1: nums = [3, -2, 1, 4], l = 2, r = 3 -> Expected: 1
    n1, l1, r1 = [3, -2, 1, 4], 2, 3
    print(f"Result 1: {sol.minimumSumSubarray(n1, l1, r1)}")

    # Example 2: nums = [-2, 2, -3, 1], l = 2, r = 3 -> Expected: -1
    n2, l2, r2 = [-2, 2, -3, 1], 2, 3
    print(f"Result 2: {sol.minimumSumSubarray(n2, l2, r2)}")

    # Example 3: nums = [1, 2, 3, 4], l = 2, r = 4 -> Expected: 3
    n3, l3, r3 = [1, 2, 3, 4], 2, 4
    print(f"Result 3: {sol.minimumSumSubarray(n3, l3, r3)}")
