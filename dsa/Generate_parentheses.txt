# Problem: https://leetcode.com/problems/generate-parentheses/
# Leetcode Problem No: 22
# Relevance: String, Dynamic Programming, Backtracking

from typing import List

class Solution:

    # Logic: Backtracking / Depth-First Search (O(4^n / sqrt(n)) Time, O(n) Auxiliary Space)
    # 1. We want to generate all combinations of well-formed parentheses given `n` pairs.
    # 2. A string of parentheses is well-formed if:
    #    - The total number of '(' equals `n` and ')' equals `n`.
    #    - At any point during construction, the number of ')' does not exceed '('.
    # 3. Recursive Backtracking State:
    #    - `open_`: Count of opening brackets placed so far.
    #    - `close_`: Count of closing brackets placed so far.
    #    - `tmp`: List of characters currently accumulated.
    # 4. Transitions & Pruning:
    #    - Base Case: If `open_ == n` and `close_ == n`, we have placed all 2*n characters.
    #      Join `tmp` into a string and append to `res`.
    #    - Choice 1: Place '(' if `open_ < n`. Recurse with `open_ + 1`, then backtrack (pop).
    #    - Choice 2: Place ')' if `close_ < open_`. Recurse with `close_ + 1`, then backtrack (pop).
    #
    # Time Complexity: O(4^n / sqrt(n)) - The number of valid parentheses sequences is the n-th Catalan number
    #                  C_n = (1 / (n + 1)) * (2n choose n) ~ O(4^n / (n^(3/2))). Generating each of length 2n takes O(n).
    # Space Complexity: O(n) auxiliary space for recursion stack and `tmp` list (excluding the output list).
    #
    # Dry run for n = 2:
    # - helper(0, 0, []):
    #   - open_ < 2 -> tmp = ['('], helper(1, 0, ['('])
    #     - open_ < 2 -> tmp = ['(', '('], helper(2, 0, ...)
    #       - close_ < 2 -> tmp = ['(', '(', ')'], helper(2, 1, ...)
    #         - close_ < 2 -> tmp = ['(', '(', ')', ')'], helper(2, 2, ...)
    #           - open_ == 2 and close_ == 2 -> res.append("(())")
    #     - close_ < 1 -> tmp = ['(', ')'], helper(1, 1, ...)
    #       - open_ < 2 -> tmp = ['(', ')', '('], helper(2, 1, ...)
    #         - close_ < 2 -> tmp = ['(', ')', '(', ')'], helper(2, 2, ...)
    #           - open_ == 2 and close_ == 2 -> res.append("()()")
    # - Result: ["(())", "()()"]

    def generateParenthesis(self, n: int) -> List[str]:
        res = []

        def helper(open_: int, close_: int, tmp: List[str]) -> None:
            if open_ == n and close_ == n:
                res.append("".join(tmp))
                return

            if open_ < n:
                tmp.append("(")
                helper(open_ + 1, close_, tmp)
                tmp.pop()

            if close_ < open_:
                tmp.append(")")
                helper(open_, close_ + 1, tmp)
                tmp.pop()

        helper(0, 0, [])
        return res

    # Example 1: n = 3 -> Expected: ["((()))","(()())","(())()","()(())","()()()"]
    n1 = 3
    print(f"Result for n = {n1}: {sol.generateParenthesis(n1)}")

    # Example 2: n = 1 -> Expected: ["()"]
    n2 = 1
    print(f"Result for n = {n2}: {sol.generateParenthesis(n2)}")

    # Example 3: n = 2 -> Expected: ["(())","()()"]
    n3 = 2
    print(f"Result for n = {n3}: {sol.generateParenthesis(n3)}")
