# Problem: https://leetcode.com/problems/maximum-k-to-sort-a-permutation/
# Leetcode Problem No: 3644
# Relevance: Array, Math, Greedy, Bit Manipulation

from typing import List

class Solution:

    # Logic: Bitwise AND Across Misplaced Elements (O(N) Time, O(1) Space)
    # 1. We are given a permutation `nums` of numbers from 0 to n - 1.
    #    We can swap indices i and j only if `nums[i] & nums[j] == k`.
    #    We want to find the maximum possible non-negative integer `k` such that `nums` can be sorted.
    # 2. Key Insights:
    #    - If an element is already in place (`nums[i] == i`), it does not necessarily need to be moved.
    #    - For any misplaced element (`nums[i] != i`), it must participate in at least one swap to reach its
    #      sorted position.
    #    - If two elements `u` and `v` are swapped, the condition requires `u & v == k`.
    #      This implies that all 1-bits in `k` must be set in both `u` and `v` (i.e. `k` is a submask of both).
    #    - By transitivity of connectivity in the swap graph, every element in a connected component that requires
    #      reordering must share all 1-bits of `k`.
    #    - Hence, `k` must be a submask of every misplaced value. The maximum such `k` is the bitwise AND
    #      of all numbers `nums[i]` where `nums[i] != i`.
    # 3. If the array is already sorted, no swaps are needed; return 0.
    #
    # Time Complexity: O(N) where N is the length of `nums`, scanning the permutation once.
    # Space Complexity: O(1) auxiliary space.
    #
    # Dry run for nums = [0, 3, 2, 1]:
    # - i = 0: nums[0] = 0 == 0 (in place)
    # - i = 1: nums[1] = 3 != 1 -> res = -1 & 3 = 3
    # - i = 2: nums[2] = 2 == 2 (in place)
    # - i = 3: nums[3] = 1 != 3 -> res = 3 & 1 = 1
    # - Final res = 1

    def sortPermutation(self, nums: List[int]) -> int:
        n = len(nums)
        res = -1
        for i in range(n):
            if nums[i] != i:
                res = res & nums[i]
        return 0 if res == -1 else res

if __name__ == "__main__":
    sol = Solution()

    # Example 1: nums = [0, 3, 2, 1] -> Expected: 1
    nums1 = [0, 3, 2, 1]
    print(f"Result for {nums1}: {sol.sortPermutation(nums1)}")

    # Example 2: nums = [0, 1, 2] -> Expected: 0
    nums2 = [0, 1, 2]
    print(f"Result for {nums2}: {sol.sortPermutation(nums2)}")

    # Example 3: nums = [3, 2, 1, 0] -> Expected: 0
    nums3 = [3, 2, 1, 0]
    print(f"Result for {nums3}: {sol.sortPermutation(nums3)}")
