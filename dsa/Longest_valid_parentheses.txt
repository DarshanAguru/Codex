# Date: 05/10/2026
# Problem: https://leetcode.com/problems/longest-valid-parentheses/
# Leetcode Problem No: 32
# Relevance: String, Dynamic Programming, Stack

class Solution:

    # Logic: Stack-Based Index Tracking (O(N) Time, O(N) Space)
    # 1. We use a stack `st` to keep track of the indices of characters.
    # 2. We initialize `st = [-1]` to serve as a base index/boundary before the start of any valid substring.
    # 3. Iterate through string `s` with index `i`:
    #    - If `s[i] == '('`:
    #      Push index `i` onto `st`.
    #    - If `s[i] == ')'`:
    #      Pop the top index from `st`.
    #      - If `st` becomes empty, this `')'` is unmatched and has no corresponding `'('`.
    #        It now serves as the new base boundary for any subsequent valid substrings, so push `i` to `st`.
    #      - If `st` is not empty, a valid parentheses substring has just been completed from `st[-1] + 1` to `i`.
    #        Its length is `i - st[-1]`. Update `res = max(res, i - st[-1])`.
    # 4. Return `res`.
    #
    # Time Complexity: O(N) where N is the length of `s`, as each index is pushed and popped at most once.
    # Space Complexity: O(N) auxiliary space for the stack.
    #
    # Dry run for s = ")()())":
    # - Init: st = [-1], res = 0
    # - i = 0, ')': pop -1 -> st empty -> st.append(0) -> st = [0]
    # - i = 1, '(': push 1 -> st = [0, 1]
    # - i = 2, ')': pop 1 -> st = [0] -> res = max(0, 2 - 0) = 2
    # - i = 3, '(': push 3 -> st = [0, 3]
    # - i = 4, ')': pop 3 -> st = [0] -> res = max(2, 4 - 0) = 4
    # - i = 5, ')': pop 0 -> st empty -> st.append(5) -> st = [5]
    # - Final: res = 4

    def longestValidParentheses(self, s: str) -> int:
        n = len(s)
        res = 0
        st = [-1]

        for i in range(n):
            if s[i] == "(":
                st.append(i)
            else:
                st.pop()
                if not st:
                    st.append(i)
                else:
                    res = max(res, i - st[-1])

        return res

    # Example 1: s = "(()" -> Expected: 2
    s1 = "(()"
    print(f"Result for '{s1}': {sol.longestValidParentheses(s1)}")

    # Example 2: s = ")()())" -> Expected: 4
    s2 = ")()())"
    print(f"Result for '{s2}': {sol.longestValidParentheses(s2)}")

    # Example 3: s = "" -> Expected: 0
    s3 = ""
    print(f"Result for '{s3}': {sol.longestValidParentheses(s3)}")
