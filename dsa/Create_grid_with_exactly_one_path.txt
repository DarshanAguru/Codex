# Problem: https://leetcode.com/problems/create-grid-with-exactly-one-path/
# Leetcode Problem No: 3963
# Relevance: Matrix, Constructive Algorithms

from typing import List

class Solution:

    # Logic: Constructive Path Isolation (O(M * N) Time, O(M * N) Space)
    # 1. We must construct an M x N grid consisting of '.' (open cells) and '#' (obstacle cells)
    #    such that there is EXACTLY ONE path moving only Right or Down from (0, 0) to (m - 1, n - 1).
    # 2. Construction Strategy:
    #    - Open the entire first row: row 0 is "." * n.
    #    - For every subsequent row (from 1 to m - 1), place obstacles at every column except the last:
    #      "#" * (n - 1) + ".".
    # 3. Path Uniqueness:
    #    - From (0, 0), moving down immediately hits '#' at (1, 0) (for m > 1).
    #    - In fact, at any column j < n - 1, moving down from row 0 is blocked by '#'.
    #    - Therefore, the path is forced to travel all the way right along row 0 to (0, n - 1).
    #    - From (0, n - 1), the only open cells in subsequent rows are strictly at column n - 1.
    #    - Hence, the path is forced to move straight down along the last column to (m - 1, n - 1).
    #    - This guarantees exactly 1 valid path from start to finish.
    #
    # Time Complexity: O(M * N) to construct the grid strings.
    # Space Complexity: O(M * N) space to store and return the grid.
    #
    # Dry run for m = 3, n = 3:
    # - Row 0: "..."
    # - Row 1: "##."
    # - Row 2: "##."
    # - Result: ["...", "##.", "##."]
    # - Valid paths: (0, 0) -> (0, 1) -> (0, 2) -> (1, 2) -> (2, 2). Exactly 1 path.

    def createGrid(self, m: int, n: int) -> List[str]:
        return ["." * n] + ["#" * (n - 1) + "."] * (m - 1)

if __name__ == "__main__":
    sol = Solution()

    # Example 1: m = 2, n = 3
    print("Result for m = 2, n = 3:")
    for row in sol.createGrid(2, 3):
        print(f"  {row}")

    # Example 2: m = 3, n = 3
    print("\nResult for m = 3, n = 3:")
    for row in sol.createGrid(3, 3):
        print(f"  {row}")

    # Example 3: m = 1, n = 1
    print("\nResult for m = 1, n = 1:")
    for row in sol.createGrid(1, 1):
        print(f"  {row}")
