# Date: 09/10/2026
# Problem: https://leetcode.com/problems/minimum-insertions-to-balance-a-parentheses-string/
# Leetcode Problem No: 1541
# Relevance: String, Stack, Greedy

class Solution:

    # Logic: Greedy Balance of Needed Closing Parentheses (O(N) Time, O(1) Space)
    # 1. In this problem, every '(' must match with exactly two consecutive ')' (i.e. "))").
    # 2. We maintain two variables:
    #    - `need`: the number of right parentheses ')' currently needed.
    #    - `res`: the number of insertions added so far.
    # 3. Iterate through string `s`:
    #    - When encountering '(':
    #      - Each '(' requires 2 closing brackets.
    #      - If `need % 2 == 1`, the previous '(' was only followed by one ')' and is still missing one ')'.
    #        Since a new '(' is beginning, we must greedily close the previous '(' by inserting one ')'
    #        (`res += 1`, `need -= 1`).
    #      - Then add 2 to `need` for the current '(' (`need += 2`).
    #    - When encountering ')':
    #      - One required ')' is satisfied: `need -= 1`.
    #      - If `need < 0`: we received an unexpected ')' without any preceding '('.
    #        We insert one '(' before it (`res += 1`). Inserting '(' requires two ')', and since we already
    #        have this current ')', we now need 1 more ')' (`need = 1`).
    # 4. At the end, any remaining `need` right brackets must be inserted, so return `res + need`.
    #
    # Time Complexity: O(N) where N is the length of string `s`, scanning each character once.
    # Space Complexity: O(1) auxiliary space using two integer variables.
    #
    # Dry run for s = "(()))":
    # - ch = '(': need += 2 -> need = 2, res = 0
    # - ch = '(': need += 2 -> need = 4, res = 0
    # - ch = ')': need -= 1 -> need = 3, res = 0
    # - ch = ')': need -= 1 -> need = 2, res = 0
    # - ch = ')': need -= 1 -> need = 1, res = 0
    # - Loop ends: return res + need = 0 + 1 = 1.

    def minInsertions(self, s: str) -> int:
        res = 0
        need = 0

        for ch in s:
            if ch == "(":
                if need % 2 == 1:
                    res += 1
                    need -= 1
                need += 2
            else:
                need -= 1
                if need < 0:
                    res += 1
                    need = 1

        return res + need

if __name__ == "__main__":
    sol = Solution()

    # Example 1: s = "(()))" -> Expected: 1
    s1 = "(()))"
    print(f"Result for '{s1}': {sol.minInsertions(s1)}")

    # Example 2: s = "())" -> Expected: 0
    s2 = "())"
    print(f"Result for '{s2}': {sol.minInsertions(s2)}")

    # Example 3: s = "))())(" -> Expected: 3
    s3 = "))())("
    print(f"Result for '{s3}': {sol.minInsertions(s3)}")
