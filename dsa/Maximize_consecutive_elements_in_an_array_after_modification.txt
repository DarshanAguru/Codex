# Problem: https://leetcode.com/problems/maximize-consecutive-elements-in-an-array-after-modification/
# Leetcode Problem No: 3041
# Relevance: Array, Dynamic Programming, Sorting

from typing import List

class Solution:

    # Logic: Sorting + Hash Map Dynamic Programming (O(N log N) Time, O(N) Space)
    # 1. We are given an integer array `nums` where each element can either remain `x` or become `x + 1`.
    #    We want to find the maximum length of a strictly consecutive sequence formed by a subset of modified elements.
    # 2. Key Insights:
    #    - Sorting `nums` ascending allows building consecutive sequences from smaller to larger numbers.
    #    - Let `dp[v]` be the length of the longest consecutive sequence ending with value `v`.
    # 3. Transitions for each element `val`:
    #    - Case 1 (promote `val` to `val + 1`):
    #      Extends a consecutive sequence ending at `val`:
    #      `dp[val + 1] = dp.get(val, 0) + 1`
    #    - Case 2 (keep `val` as `val`):
    #      Extends a consecutive sequence ending at `val - 1`:
    #      `dp[val] = dp.get(val - 1, 0) + 1`
    #    - Notice the order: We update `dp[val + 1]` first using the pre-existing value of `dp[val]`
    #      to avoid using the current element twice in the same chain.
    # 4. Track the maximum value observed in `dp` throughout iterations.
    #
    # Time Complexity: O(N log N) where N is the length of `nums`, dominated by sorting.
    # Space Complexity: O(N) auxiliary space to store DP states in the hash map.
    #
    # Dry run for nums = [2, 1, 5, 1, 1]:
    # - Sorted: [1, 1, 1, 2, 5]
    # - 1st '1': dp[2] = dp[1] + 1 = 1; dp[1] = dp[0] + 1 = 1 (max = 1)
    # - 2nd '1': dp[2] = dp[1] + 1 = 2; dp[1] = dp[0] + 1 = 1 (max = 2)
    # - 3rd '1': dp[2] = dp[1] + 1 = 2; dp[1] = dp[0] + 1 = 1 (max = 2)
    # - '2': dp[3] = dp[2] + 1 = 3; dp[2] = dp[1] + 1 = 2 (max = 3)
    # - '5': dp[6] = 1, dp[5] = 1 (max = 3)
    # - Return: 3 (sequence: [1, 2, 3]).

    def maxSelectedElements(self, nums: List[int]) -> int:
        nums.sort()
        dp = {}
        res = 0

        for val in nums:
            dp[val + 1] = dp.get(val, 0) + 1
            dp[val] = dp.get(val - 1, 0) + 1
            res = max(res, dp[val], dp[val + 1])

        return res

if __name__ == "__main__":
    sol = Solution()

    # Example 1: nums = [2, 1, 5, 1, 1] -> Expected: 3
    nums1 = [2, 1, 5, 1, 1]
    print(f"Result 1: {sol.maxSelectedElements(nums1)}")

    # Example 2: nums = [1, 4, 7, 10] -> Expected: 1
    nums2 = [1, 4, 7, 10]
    print(f"Result 2: {sol.maxSelectedElements(nums2)}")

    # Example 3: nums = [8, 10, 6, 12, 9, 12, 2, 3, 13, 19, 11, 18, 10, 16] -> Expected: 8
    nums3 = [8, 10, 6, 12, 9, 12, 2, 3, 13, 19, 11, 18, 10, 16]
    print(f"Result 3: {sol.maxSelectedElements(nums3)}")
