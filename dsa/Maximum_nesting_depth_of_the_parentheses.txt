# Relevance: String, Stack

class Solution:

    # Logic: Counter / Stack-Based Depth Tracking (O(N) Time, O(1) Space)
    # 1. A string `s` is a valid parentheses string (VPS). We want to find its maximum nesting depth.
    # 2. As we iterate through each character in `s`:
    #    - When encountering an opening parenthesis `'('`, the current nesting depth increases by 1 (`_open += 1`).
    #    - When encountering a closing parenthesis `')'`, the current nesting depth decreases by 1 (`_open -= 1`).
    #    - Other characters (digits, operators) do not affect the depth.
    # 3. At each step, update the maximum depth seen so far: `res = max(res, _open)`.
    # 4. Return `res`.
    #
    # Time Complexity: O(N) where N is the length of string `s`.
    # Space Complexity: O(1) auxiliary space as we only use counter variables.
    #
    # Dry run for s = "(1+(2*3)+((8)/4))+1":
    # - '(' -> _open = 1, res = 1
    # - '1', '+', '(' -> _open = 2, res = 2
    # - '2', '*', '3', ')' -> _open = 1, res = 2
    # - '+', '(', '(' -> _open = 3, res = 3
    # - '8', ')', '/', '4', ')' -> _open = 1, res = 3
    # - ')' -> _open = 0, res = 3
    # - '+', '1' -> _open = 0, res = 3
    # - Returns 3.

    def maxDepth(self, s: str) -> int:
        res = 0
        _open = 0

        for c in s:
            if c == "(":
                _open += 1
            elif c == ")":
                _open -= 1
            res = max(res, _open)

        return res
