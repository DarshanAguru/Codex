# Problem: https://leetcode.com/problems/valid-parentheses/
# Leetcode Problem No: 20
# Relevance: String, Stack, Bit Manipulation

class Solution:

    # Logic: Bit-Level Stack Simulation Using Integer Bitwise Shifts (O(N) Time, O(1) / O(N) Bits Space)
    # 1. Instead of a traditional list stack, we use an integer `mask` to simulate a stack of brackets:
    #    - Each bracket is uniquely encoded using 3 bits:
    #      '(' -> 1 (0b001)
    #      '[' -> 2 (0b010)
    #      '{' -> 4 (0b100)
    # 2. Operations:
    #    - Push: Shift `mask` left by 3 bits and OR with the bracket's value: `mask = (mask << 3) | openMap[c]`.
    #    - Peek: Extract the lowest 3 bits using bitwise AND: `top = mask & 7`.
    #    - Pop: Shift `mask` right by 3 bits: `mask = mask >> 3`.
    # 3. For each character in `s`:
    #    - If it is an opening bracket (`'(', '[', '{'`), push it to `mask`.
    #    - If it is a closing bracket (`')', ']', '}'`):
    #      - If `mask == 0`, there is no matching opening bracket -> return `False`.
    #      - Peek the top 3 bits. If they don't match `closeMap[c]`, mismatched bracket -> return `False`.
    #      - Otherwise, pop the top bracket (`mask >>= 3`).
    # 4. At the end, return `True` if `mask == 0` (all brackets were matched and popped), else `False`.
    #
    # Time Complexity: O(N) where N is the length of string `s`.
    # Space Complexity: O(1) auxiliary space (Python dynamically handles large integers in memory).
    #
    # Dry run for s = "([])":
    # - '(': mask = (0 << 3) | 1 = 1 (0b001)
    # - '[': mask = (1 << 3) | 2 = 10 (0b001010)
    # - ']': top = 10 & 7 = 2. Matches closeMap[']'] (2). Pop: mask = 10 >> 3 = 1 (0b001)
    # - ')': top = 1 & 7 = 1. Matches closeMap[')'] (1). Pop: mask = 1 >> 3 = 0
    # - Final: mask == 0 -> Returns True.

    def isValid(self, s: str) -> bool:
        mask = 0
        openMap = {"(": 1, "[": 2, "{": 4}
        closeMap = {")": 1, "]": 2, "}": 4}

        for c in s:
            if c in "([{":
                mask = (mask << 3) | openMap[c]
            else:
                if mask == 0:
                    return False

                top = mask & 7

                if top != closeMap[c]:
                    return False

                mask = mask >> 3

        return mask == 0

    s5 = "([)]"
